Показать защищённый SQL для Supabase (RLS + аккаунты + списки) 
create table if not exists public.ranoa_rooms (
  room_id text primary key check (char_length(room_id) between 3 and 64),
  items jsonb not null default '[]'::jsonb check (jsonb_typeof(items) = 'array'),
  custom_tags jsonb not null default '[]'::jsonb check (jsonb_typeof(custom_tags) = 'array'),
  shared_lists jsonb not null default '[]'::jsonb check (jsonb_typeof(shared_lists) = 'array'),
  accounts jsonb not null default '[]'::jsonb check (jsonb_typeof(accounts) = 'array'),
  updated_at timestamptz not null default now()
);
alter table public.ranoa_rooms add column if not exists shared_lists jsonb not null default '[]'::jsonb;
alter table public.ranoa_rooms add column if not exists accounts jsonb not null default '[]'::jsonb;
alter table public.ranoa_rooms enable row level security;
drop policy if exists "Public room access" on public.ranoa_rooms;
drop policy if exists "Scoped room read" on public.ranoa_rooms;
drop policy if exists "Scoped room insert" on public.ranoa_rooms;
drop policy if exists "Scoped room update" on public.ranoa_rooms;
create policy "Scoped room read" on public.ranoa_rooms
  for select using (char_length(room_id) >= 4);
create policy "Scoped room insert" on public.ranoa_rooms
  for insert with check (char_length(room_id) >= 4 and jsonb_array_length(items) <= 1000);
create policy "Scoped room update" on public.ranoa_rooms
  for update using (char_length(room_id) >= 4) with check (jsonb_array_length(items) <= 1000);
do $$
begin
  if not exists (
    select 1 from pg_publication_tables
    where pubname = 'supabase_realtime' and schemaname = 'public' and tablename = 'ranoa_rooms'
  ) then
    alter publication supabase_realtime add table public.ranoa_rooms;
  end if;
end $$;
